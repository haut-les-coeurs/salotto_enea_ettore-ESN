\newcommand{\decoframe}[4]{% inset, smusso, colore, spessore
 \draw[#3,line width=#4] (#1+#2,#1) -- (\paperwidth-#1-#2,#1) -- (\paperwidth-#1,#1+#2)
   -- (\paperwidth-#1,\paperheight-#1-#2) -- (\paperwidth-#1-#2,\paperheight-#1)

\newcommand{\cornerfan}[4]{% x,y,rotazione,colore
 \begin{scope}[shift={(#1,#2)},rotate=#3]
  \foreach \a in {0,15,...,90}{\draw[#4,line width=.3pt] (0,0) -- (\a:8mm);}
  \draw[#4,line width=.3pt] (8mm,0) arc(0:90:8mm);
  \draw[#4,line width=.3pt] (5mm,0) arc(0:90:5mm);
 \end{scope}}

\newcommand{\cardnode}[4]{%
	\edef\c@rank{#1}\expandafter\let\expandafter\c@suit\csname suit@#2\endcsname
	\expandafter\let\expandafter\c@col\csname scol@#2\endcsname
	\edef\c@mode{#4}\def\c@u{u}\def\c@d{d}%
	\begin{scope}[shift={(#3,0)}]
		\ifx\c@mode\c@u\begin{scope}[yshift=2.6mm]\fi
			\ifx\c@rank\x@back
			\draw[fill=ink,draw=gold,line width=.5pt,rounded corners=.9mm] (0,0) rectangle (\cw,{1.4*\cw});
			\draw[gold,line width=.3pt,rounded corners=.4mm] (1.2mm,1.2mm) rectangle ({\cw-1.2mm},{1.4*\cw-1.2mm});
			\foreach \a in {20,50,...,160}{\draw[gold,line width=.3pt] ({.5*\cw},{.45*\cw}) -- +(\a:6mm);}
			\node[diamond,fill=gold,inner sep=1.4pt] at ({.5*\cw},{.7*\cw}) {};
			\else
			\draw[fill=white,draw=ink!65,line width=.3pt,rounded corners=.9mm] (0,0) rectangle (\cw,{1.4*\cw});
			\ifx\c@u\c@mode\draw[gold,line width=1.1pt,rounded corners=.9mm] (0,0) rectangle (\cw,{1.4*\cw});\fi
			\def\c@z{Z}\edef\c@s{#2}
			\ifx\c@s\c@z
			\node at ({.5*\cw},{.7*\cw}) {\suitfont\fontsize{20}{20}\selectfont\color{\c@col}\c@suit};
			\node[anchor=north west,inner sep=0] at (1mm,{1.4*\cw-.5mm}) {\fontsize{5.2}{5.2}\selectfont\bfseries\color{rose}JOLLY};
			\else
			% --- Angolo in alto a sinistra ---
			\node[anchor=north,inner sep=0] at (1.9mm,{1.4*\cw-.6mm}) 
			{\fontsize{9}{9}\selectfont\bfseries\color{\c@col}\c@rank};
			\node[anchor=north,inner sep=0] at (1.9mm,{1.4*\cw-3.5mm}) 
			{\suitfont\fontsize{6.5}{6.5}\selectfont\color{\c@col}\c@suit};
			
			% --- Angolo in basso a destra ---
			\node[anchor=north,inner sep=0,rotate=180] at ({\cw-1.9mm},.6mm) 
			{\fontsize{9}{9}\selectfont\bfseries\color{\c@col}\c@rank};
			\node[anchor=north,inner sep=0,rotate=180] at ({\cw-1.9mm},{3.5mm}) 
			{\suitfont\fontsize{6.5}{6.5}\selectfont\color{\c@col}\c@suit};
			
			% --- Simbolo o Figura centrale (pulita) ---
			\def\c@J{J}\def\c@Q{Q}\def\c@K{K}
			\ifx\c@rank\c@J\def\c@face{1}\else\ifx\c@rank\c@Q\def\c@face{1}\else\ifx\c@rank\c@K\def\c@face{1}\else\def\c@face{0}\fi\fi\fi
			\if\c@face1
			\node at ({.5*\cw},{.7*\cw}) {\bfseries\fontsize{15}{15}\selectfont\color{\c@col}\c@rank};
			\else
			\node at ({.5*\cw},{.7*\cw}) {\suitfont\fontsize{16}{16}\selectfont\color{\c@col}\c@suit};
			\fi
			\fi
			\fi
			\ifx\c@mode\c@d\fill[cream,opacity=.68,rounded corners=.9mm] (0,0) rectangle (\cw,{1.4*\cw});\fi
			\ifx\c@mode\c@u\end{scope}\fi
\end{scope}}
